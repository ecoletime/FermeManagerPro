  if(regle&&!regle.active)return;
  const d=detail||'—';
  const notif={
    id:nextNotifId++,
    type:tmpl.type,
    cat:tmpl.cat,
    icon:tmpl.icon,
    title:tmpl.title.replace('{detail}',d),
    desc:tmpl.desc.replace(/{detail}/g,d),
    time:now(),

function addDestinataire(){
  const nom=document.getElementById('dest-nom').value.trim();
  const email=document.getElementById('dest-email').value.trim();
  if(!nom||!email){alert('Nom et email requis.');return;}
  if(!email.includes('@')){alert('Email invalide.');return;}
  destinataires.push({
    id:nextDestId++,nom,email,role:document.getElementById('dest-role').value,
    notifs:{
      critique:document.getElementById('dn-critique').checked,
      alerte:document.getElementById('dn-alerte').checked,
      info:document.getElementById('dn-info').checked,
      resume:document.getElementById('dn-resume').checked,
    },
    actif:true,
  });
  showAppToast('"'+nom+'" ajouté comme destinataire','g');
  ['dest-nom','dest-email','dest-tel'].forEach(id=>document.getElementById(id).value='');
  renderDestinatairesList();renderDashboard();
}
