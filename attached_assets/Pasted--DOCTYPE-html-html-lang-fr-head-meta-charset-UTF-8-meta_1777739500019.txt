function terminerMaint(id){
  const m=maintenances.find(x=>x.id===id);if(!m)return;
  if(CU&&CU.role!=='admin'&&m.saisie!==CU.nom){toast('Seul l\'admin peut modifier les tâches des autres','red');return;}
  m.statut='termine';toast('"'+m.titre+'" terminée','green');renderMaint();
}
function delMaint(id){
  if(CU&&CU.role!=='admin'){toast('Suppression réservée à l\'administrateur','red');return;}
  if(!confirm('Supprimer cette tâche ?'))return;
  maintenances=maintenances.filter(x=>x.id!==id);toast('Tâche supprimée','amber');renderMaint();
}
function openEditMaint(id){

  document.getElementById('maint-edit-title').textContent='Modifier — '+m.titre;

}
